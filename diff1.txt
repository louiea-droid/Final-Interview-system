+/*
+ * How an element's stored geometry becomes CSS.
+ *
+ * Sizes are expressed in container query units against the board's own box -
+ * cqw for widths, cqh for type - so the live board (a full viewport) and the
+ * editor's canvas (a 16:9 panel) lay elements out identically. Both mark
+ * themselves as the query container; everything here is relative to that.
+ */
+
+/* Logo and photo widths at 100%, as a share of the board's width. */
+export const LOGO_BASE_WIDTH = 20;
+export const PHOTO_BASE_WIDTH = 11.5;
+
+/** Centres an element on its stored percentage position. */
+export function positionStyle(element) {
+  return {
+    left: `${element.x}%`,
+    top: `${element.y}%`,
+    transform: 'translate(-50%, -50%)',
+  };
+}
+
+/** The same, for elements whose size is a multiplier of a base width. */
+export function scaledPositionStyle(element, baseWidth) {
+  return {
+    left: `${element.x}%`,
+    top: `${element.y}%`,
+    transform: `translate(-50%, -50%) scale(${element.scale})`,
+    width: `${baseWidth}cqw`,
+  };
+}

+/*
+ * Free-form layout of the elements on /visual - the logo, every text block,
+ * any photos, and any shapes or photo frames the admin drops in. Positions
+ * are percentages of the board's width/height and type is sized against its
+ * height, so a layout holds up on any display; each element carries its own
+ * `visible` flag so one can be turned off without losing its place.
+ *
+ * The headline and the closing line are ordinary entries in `texts` - they
+ * start the board off, but they can be restyled, moved, or removed like any
+ * other text.
+ *
+ * Editing happens on a separate draft copy (its own localStorage key) so
+ * the public board never shows work in progress - /visual only ever reads
+ * the published copy, and stays completely unaware that a draft exists.
+ * Publishing copies the draft over it and notifies other tabs the same way
+ * interview-board-settings does: a 'storage' event reaches them on its own,
+ * and LAYOUT_EVENT covers a listener in the same window.
+ */
+

+  DEFAULT_CLOSING_STYLE,
+  DEFAULT_EXTRA_TEXT_STYLE,

+  DEFAULT_TEXT_SIZES,
+  normalizeTextStyle,

+import { normalizeShapes } from './shapeStyle';

+/*
+ * Bumped when stored geometry stops meaning what it used to. Positions were
+ * once percentages of a middle column that took about half the board's
+ * width; on one free-form surface those same numbers point somewhere else
+ * entirely, so a layout from before the change is dropped rather than
+ * reinterpreted into a scramble.
+ */

+
+export const LAYOUT_KEY = 'interview-board-layout';
+export const DRAFT_KEY = 'interview-board-layout:draft';
+export const LAYOUT_EVENT = 'interview-board-layout-updated';

+function clamp(value, min, max) {
+  return Math.min(max, Math.max(min, Number.isFinite(value) ? value : min));
+}
+
+function normalizeText(text, base = DEFAULT_EXTRA_TEXT_STYLE, baseSize = DEFAULT_TEXT_SIZES.extra) {
+  return {
+    id: text.id ?? crypto.randomUUID(),
+    text: typeof text.text === 'string' ? text.text : '',
+    x: clamp(text.x, 0, 100),
+    y: clamp(text.y, 0, 100),
+    // `scale` was a multiplier of a per-role base size before sizes were stored outright
+    fontSize: Number.isFinite(text.fontSize)
+      ? clamp(text.fontSize, 0.3, 40)
+      : clamp(baseSize * (Number(text.scale) || 1), 0.3, 40),
+    visible: text.visible !== false,
+    /* How wide the text may run before wrapping, as a share of the board. */
+    boxWidth: clamp(text.boxWidth ?? 92, 5, 100),
+    style: normalizeTextStyle(text.style, base),
+  };
+}
+
+function layerKey(entry) {
+  return `${entry.kind}:${entry.id ?? ''}`;
+}
+
+/*
+ * Rebuilds the paint order against the elements that actually exist: drops
+ * any entry naming something that was deleted, then appends the logo, the
+ * candidate board, and any element (old data, or one added after the saved
+ * order) that isn't in it yet - new things paint on top, which is where
+ * they'd have landed had layerOrder always existed.
+ */
+function normalizeLayerOrder(saved, layout) {
+  const real = [
+    { kind: 'logo', id: null },
+    { kind: 'candidates', id: null },
+    { kind: 'announcements', id: null },
+    ...layout.texts.map((item) => ({ kind: 'text', id: item.id })),
+    ...layout.dividers.map((item) => ({ kind: 'divider', id: item.id })),
+    ...layout.images.map((item) => ({ kind: 'image', id: item.id })),
+    ...layout.shapes.map((item) => ({ kind: 'shape', id: item.id })),
+  ];
+  const realKeys = new Set(real.map(layerKey));
+
+  const ordered = Array.isArray(saved)
+    ? saved.filter((entry) => entry && realKeys.has(layerKey(entry)))
+    : [];
+
+  const seen = new Set(ordered.map(layerKey));
+
+  for (const entry of real) {
+    if (!seen.has(layerKey(entry))) {
+      ordered.push(entry);
+      seen.add(layerKey(entry));
+    }
+  }
+
+  return ordered;
+}
+
+/*
+ * Every divider on the board. A layout from when there was only ever one
+ * carries it over as the first entry of the list.
+ */
+function normalizeDividers(parsed) {
+  const source = Array.isArray(parsed.dividers)
+    ? parsed.dividers
+    : parsed.divider
+      ? [parsed.divider]
+      : [];
+
+  return source
+    .filter((item) => item && typeof item === 'object')
+    .map((item) => ({
+      id: item.id ?? crypto.randomUUID(),
+      x: clamp(item.x ?? DEFAULT_DIVIDER.x, 0, 100),
+      y: clamp(item.y ?? DEFAULT_DIVIDER.y, 0, 100),
+      // `height` was the field before dividers could also run across
+      length: clamp(item.length ?? item.height ?? DEFAULT_DIVIDER.length, 2, 100),
+      thickness: clamp(item.thickness ?? DEFAULT_DIVIDER.thickness, 1, 24),
+      orientation: item.orientation === 'horizontal' ? 'horizontal' : 'vertical',
+      visible: item.visible !== false,
+      locked: item.locked === true,
+      ...(item.groupId ? { groupId: item.groupId } : {}),
+    }));
+}
+
+/* A positioned, sized box - the candidate board and the announcement strip. */
+function normalizeBox(box, defaults) {
+  const source = box && typeof box === 'object' ? box : {};
+
+  return {
+    visible: source.visible !== false,
+    locked: source.locked === true,
+    x: clamp(source.x ?? defaults.x, 0, 100),
+    y: clamp(source.y ?? defaults.y, 0, 100),
+    ...(defaults.width === undefined
+      ? {}
+      : { width: clamp(source.width ?? defaults.width, 1, 100) }),
+    ...(defaults.height === undefined
+      ? {}
+      : { height: clamp(source.height ?? defaults.height, 8, 100) }),
+    ...(defaults.secondsPerCard === undefined
+      ? {}
+      : {
+          secondsPerCard: clamp(
+            source.secondsPerCard ?? defaults.secondsPerCard,
+            0.5,
+            30
+          ),
+        }),
+  };
+}
+
+function normalizeLayout(parsed) {
+  const texts = [];
+
+  /*
+   * Layouts saved before this kept the headline and the closing line as
+   * fixed singletons; they become the first two entries of the text list,
+   * where they can now be edited and removed like any other.
+   */
+  if (parsed.heading) {
+    texts.push(
+      normalizeText(
+        { ...parsed.heading, id: 'heading' },
+        DEFAULT_HEADING_STYLE,
+        DEFAULT_TEXT_SIZES.heading
+      )
+    );
+  }
+
+  if (parsed.closing) {
+    texts.push(
+      normalizeText(
+        { ...parsed.closing, id: 'closing' },
+        DEFAULT_CLOSING_STYLE,
+        DEFAULT_TEXT_SIZES.closing
+      )
+    );
+  }
+
+  if (Array.isArray(parsed.texts)) {
+    for (const text of parsed.texts) {
+      if (text && typeof text === 'object') texts.push(normalizeText(text));
+    }
+  }
+
+  // an admin who removed every text keeps an empty board, rather than the defaults
+  const hasTextData = Boolean(parsed.heading || parsed.closing || Array.isArray(parsed.texts));
+
+  const layout = {
+    logo: { ...DEFAULT_LAYOUT.logo, ...(parsed.logo ?? {}) },
+    candidates: normalizeBox(parsed.candidates, DEFAULT_CANDIDATES),
+    announcements: normalizeBox(parsed.announcements, DEFAULT_ANNOUNCEMENTS),
+    dividers: normalizeDividers(parsed),
+    texts: hasTextData ? texts : DEFAULT_LAYOUT.texts,
+    images: Array.isArray(parsed.images) ? parsed.images : [],
+    shapes: normalizeShapes(parsed.shapes),
+  };
+
+  layout.version = LAYOUT_VERSION;
+  layout.layerOrder = normalizeLayerOrder(parsed.layerOrder, layout);
+
+  return layout;
+}
+
+function readLayout(key) {
+  try {
+    const raw = window.localStorage.getItem(key);
+    if (!raw) return null;
+
+    const parsed = JSON.parse(raw);
+
+    // geometry from before the free-form change cannot be carried over
+    if (parsed?.version !== LAYOUT_VERSION) {
+      window.localStorage.removeItem(key);
+      return null;
+    }
+
+    return normalizeLayout(parsed);
+  } catch {
+    return null;
+  }
+}

+  return element.visible !== false;

+  return readLayout(LAYOUT_KEY) ?? DEFAULT_LAYOUT;

+/**
+ * The admin's working copy. Falls back to the published layout the first
+ * time the editor is opened, so editing always starts from what is
+ * currently live rather than the bare defaults.

+export function loadDraftLayout() {
+  return readLayout(DRAFT_KEY) ?? loadBoardLayout();

+/** Saves the draft only - /visual never sees this until it's published. */
+export function saveDraftLayout(layout) {
+  try {
+    window.localStorage.setItem(DRAFT_KEY, JSON.stringify(layout));
+    return true;
+  } catch (error) {
+    console.warn('Could not save the draft board layout:', error);
+    return false;
+  }

+/** Makes `layout` the published layout /visual reads, and notifies this tab. */
+export function saveBoardLayout(layout) {
+  try {
+    window.localStorage.setItem(LAYOUT_KEY, JSON.stringify(layout));
+  } catch (error) {
+    console.warn('Could not save the board layout:', error);
+    return false;
+  }
+
+  window.dispatchEvent(new Event(LAYOUT_EVENT));
+  return true;
+}
+
+/** Publishes the current draft: it becomes the live layout /visual shows. */
+export function publishBoardLayout(layout) {
+  saveDraftLayout(layout);
+  return saveBoardLayout(layout);
+}

+/*
+ * Named snapshots of a layout, saved on the side so an admin can come back
+ * to a look they liked without it living on the live board or the draft -
+ * saving one doesn't touch either, and applying one only replaces the draft.
+ */
+
+function readTemplates() {
+  try {
+    const raw = window.localStorage.getItem(TEMPLATES_KEY);
+    const parsed = raw ? JSON.parse(raw) : [];
+    return Array.isArray(parsed) ? parsed : [];
+  } catch {
+    return [];
+  }
+}
+
+function writeTemplates(templates) {
+  try {
+    window.localStorage.setItem(TEMPLATES_KEY, JSON.stringify(templates));
+    return true;
+  } catch (error) {
+    console.warn('Could not save templates:', error);
+    return false;
+  }
+}
+
+/** Every saved template, newest first. */
+export function loadTemplates() {
+  return readTemplates();
+}
+
+/** Saves `layout` as a new named template; returns it. */
+export function saveTemplate(name, layout) {
+  const template = {
+    id: crypto.randomUUID(),
+    name: name.trim() || 'Untitled template',
+    savedAt: new Date().toISOString(),
+    layout,
+  };
+
+  writeTemplates([template, ...readTemplates()]);
+  return template;
+}
+
+/** The normalized layout of a saved template, or null if it's gone. */
+export function applyTemplate(id) {
+  const template = readTemplates().find((item) => item.id === id);
+  return template ? normalizeLayout(template.layout) : null;
+}
+
+export function deleteTemplate(id) {
+  writeTemplates(readTemplates().filter((template) => template.id !== id));

+/*
+ * Shapes and photo frames. Both are boxes on the board: positioned by the
+ * same centre percentages every other element uses, sized as a percentage of
+ * the board so they hold up at any screen size, and painted with the shared
+ * fill model (a solid colour, a gradient, or - for a frame - a photo).
+ *
+ * A frame is just a box whose fill is an image, which is why it lives here
+ * rather than in its own list: everything except its label and its starting
+ * fill behaves exactly like a rectangle.
+ */
+
+import { fillBoxCss, imageFill, normalizeFill, solidFill } from './fill';
+
+export const SHAPE_KINDS = [
+  { value: 'rectangle', label: 'Rectangle' },
+  { value: 'ellipse', label: 'Ellipse' },
+  { value: 'triangle', label: 'Triangle' },
+  { value: 'line', label: 'Line' },
+  { value: 'frame', label: 'Photo frame' },
+];
+
+export const FIT_OPTIONS = [
+  { value: 'cover', label: 'Fill' },
+  { value: 'contain', label: 'Fit' },
+];
+
+const BASE_SHAPE = {
+  kind: 'rectangle',
+  x: 50,
+  y: 50,
+  width: 22,
+  height: 14,
+  rotation: 0,
+  radius: 6,
+  stroke: { color: '#c9a03c', width: 0 },
+  visible: true,
+};
+
+/* What each kind starts out as when it is dropped on the board. */
+const SHAPE_PRESETS = {
+  rectangle: { fill: solidFill('#c9a03c', 0.85) },
+  ellipse: { width: 16, height: 16, radius: 0, fill: solidFill('#d45d59', 0.85) },
+  triangle: { width: 16, height: 14, radius: 0, fill: solidFill('#f4ca5d', 0.9) },
+  line: { width: 30, height: 0.6, radius: 999, fill: solidFill('#c9a03c', 1) },
+  frame: {
+    width: 16,
+    height: 20,
+    radius: 4,
+    fill: imageFill(),
+    stroke: { color: '#c9a03c', width: 2 },
+  },
+};
+
+export function createShape(kind = 'rectangle') {
+  const preset = SHAPE_PRESETS[kind] ?? SHAPE_PRESETS.rectangle;
+
+  return {
+    ...BASE_SHAPE,
+    ...preset,
+    kind,
+    id: crypto.randomUUID(),
+    stroke: { ...(preset.stroke ?? BASE_SHAPE.stroke) },
+  };
+}
+
+export function shapeLabel(shape) {
+  return SHAPE_KINDS.find((entry) => entry.value === shape.kind)?.label ?? 'Shape';
+}
+
+export function isFrame(shape) {
+  return shape.kind === 'frame';
+}
+
+function clamp(value, min, max) {
+  return Math.min(max, Math.max(min, Number.isFinite(value) ? value : min));
+}
+
+/* Shapes come back from storage, so every field is checked before it is used. */
+export function normalizeShapes(shapes) {
+  if (!Array.isArray(shapes)) return [];
+
+  return shapes
+    .filter((shape) => shape && typeof shape === 'object' && shape.id)
+    .map((shape) => {
+      const kind = SHAPE_KINDS.some((entry) => entry.value === shape.kind)
+        ? shape.kind
+        : 'rectangle';
+
+      return {
+        ...BASE_SHAPE,
+        ...shape,
+        kind,
+        x: clamp(shape.x, 0, 100),
+        y: clamp(shape.y, 0, 100),
+        width: clamp(shape.width, 0.2, 200),
+        height: clamp(shape.height, 0.2, 200),
+        rotation: clamp(shape.rotation ?? 0, -180, 180),
+        radius: clamp(shape.radius ?? 0, 0, 999),
+        stroke: {
+          color: shape.stroke?.color ?? BASE_SHAPE.stroke.color,
+          width: clamp(shape.stroke?.width ?? 0, 0, 40),
+        },
+        fill: normalizeFill(shape.fill, SHAPE_PRESETS[kind]?.fill ?? BASE_SHAPE.fill),
+      };
+    });
+}
+
+/** Builds the inline `style` object that draws a shape or frame. */
+export function resolveShapeStyle(shape) {
+  return {
+    left: `${shape.x}%`,
+    top: `${shape.y}%`,
+    width: `${shape.width}%`,
+    height: `${shape.height}%`,
+    transform: `translate(-50%, -50%) rotate(${shape.rotation}deg)`,
+    borderRadius: shape.kind === 'ellipse' ? '50%' : `${shape.radius}px`,
+    ...(shape.stroke.width > 0
+      ? { border: `${shape.stroke.width}px solid ${shape.stroke.color}` }
+      : {}),
+    ...(shape.kind === 'triangle'
+      ? { clipPath: 'polygon(50% 0%, 0% 100%, 100% 100%)' }
+      : {}),
+    ...fillBoxCss(shape.fill),
+  };
+}

+  /*
+   * A divider's CSS. The grab area is padded out to a usable thickness even
+   * when the rule itself is a hairline, so a 1px line is still easy to hit.
+   */
+  function dividerStyle(divider) {
+    const vertical = divider.orientation !== 'horizontal';
+    const grab = Math.max(divider.thickness, 10);
+
+    return {
+      left: `${divider.x}%`,
+      top: `${divider.y}%`,
+      width: vertical ? `${grab}px` : `${divider.length}%`,
+      height: vertical ? `${divider.length}%` : `${grab}px`,
+      transform: 'translate(-50%, -50%)',
+    };
+  }

+      width: `${box.width}%`,
+      ...(box.height === undefined ? {} : { height: `${box.height}%` }),

+    };
+  }

+function mulberry32(seed) {
+  return () => {
+    seed |= 0;
+    seed = (seed + 0x6d2b79f5) | 0;
+
+    let t = Math.imul(seed ^ (seed >>> 15), 1 | seed);
+
+    t =
+      (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^
+      t;
+
+    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
+  };

+function sparkField(
+  seed,
+  count,
+  maxR

+  const rand = mulberry32(seed);

+
+  for (let i = 0; i < count; i++) {
+    const x = Math.floor(rand() * 1600);
+    const y = Math.floor(rand() * 1000);

+    const r = +(
+      rand() * rand() * maxR

+    const blur = +(
+      (r + 0.5) *
+      (1.5 + rand() * 2)
+    ).toFixed(1);
+
+    const color =
+      SPARK_COLORS[
+        Math.floor(
+          rand() * SPARK_COLORS.length
+        )
+      ];

+    const alpha = +(
+      0.25 + rand() * 0.55
+    ).toFixed(2);
+
+    shadows.push(

+        }px rgba(${color},${+(
+          alpha * 0.35
+        ).toFixed(2)})`
+      );
+    }
+  }

+  return shadows.join(", ");

+  /* Groups a flat list into rows of `size` - unlike a flat grid, a lone
+     last candidate is marked to span the full row instead of leaving an
+     empty cell next to it. */
+  function chunkIntoRows(list, size) {
+    const rows = [];
+    for (let i = 0; i < list.length; i += size) {
+      rows.push(list.slice(i, i + size));
+    }
+    return rows;
+  }
+
+  /* One candidate box for the carousel - photo on top, gold name bar below,
+     the board's original card look, just sized to sit two across.
+     `full` spans both grid columns, for a lone candidate closing out a row. */
+  function renderCandidateCard(candidate, key, full = false) {
+    return (
+      <div
+        key={key}
+        className={`flex flex-col overflow-hidden rounded-[3px] ${full ? "col-span-2" : ""} ${lightMode ? "bg-[#fdf6f3]" : "bg-[#2e0505]"}`}
+      >
+        <div
+          className={`
+            relative
+            aspect-[11/10]
+            overflow-hidden
+            ${
+              lightMode
+                ? "bg-[radial-gradient(ellipse_70%_70%_at_50%_45%,#f6ded4_0%,#e8c3b6_45%,#d8ab9c_100%)]"
+                : "bg-[radial-gradient(ellipse_70%_70%_at_50%_45%,#a51d1d_0%,#6d1010_45%,#2e0505_100%)]"
+            }
+          `}
+        >
+          {candidate.photo_url && (
+            <div
+              className="absolute inset-0 bg-cover bg-top bg-no-repeat"
+              style={{ backgroundImage: `url("${candidate.photo_url}")` }}
+            />
+          )}
+        </div>
+
+        <div
+          className={`
+            relative
+            flex
+            min-h-[clamp(48px,16cqh,84px)]
+            flex-col
+            items-center
+            justify-center
+            gap-0.5
+            px-2
+            py-1.5
+
+            bg-[linear-gradient(to_bottom,#f2d888_0%,#dbb14b_38%,#b0821c_74%,#8b6615_100%)]
+          `}
+        >
+          <AutoFitText
+            minScale={0.55}
+            className="
+              font-magneton
+              text-center
+              leading-[1.1]
+              tracking-[0.01em]
+              text-[clamp(17px,4.6cqh,28px)]
+              text-[#560606]
+              [text-shadow:1px_1px_1px_rgba(63,39,10,.3)]
+            "
+          >
+            {candidate.name}
+          </AutoFitText>
+
+          {candidate.course && (
+            <span
+              className="
+                truncate
+                text-[clamp(8px,1.7cqh,13px)]
+                font-bold
+                uppercase
+                tracking-wider
+                text-[#2e1604]/80
+              "
+            >
+              {candidate.course}
+            </span>
+          )}
+        </div>
+      </div>
+    );
+  }
+
+  /* The right column's whole candidate board: a loading skeleton, an empty
+     message, or the looping carousel - two rows of two copies stacked so
+     translateY(-50%) always lands exactly on the seam between them. */
+  function renderCandidateCarousel() {
+    if (loading) {
+      return (
+        <div className="grid h-full w-full auto-rows-[minmax(0,1fr)] grid-cols-2 gap-[2cqw] overflow-hidden [container-type:size]">
+          {Array.from({ length: CAROUSEL_VISIBLE_COUNT }).map((_, index) => (
+            <div
+              key={`skeleton-${index}`}
+              style={{ animationDelay: `${index * 0.12}s` }}
+              className={`
+                flex
+                flex-col
+                overflow-hidden
+                animate-[skeleton-in_.5s_ease-out_both]
+                ${lightMode ? "bg-[#fdf6f3]" : "bg-[#2e0505]"}
+              `}
+            >
+              <div
+                className={`
+                  relative
+                  aspect-[11/10]
+                  animate-[pulse-glow_1.8s_ease-in-out_infinite]
+                  ${
+                    lightMode
+                      ? "bg-[radial-gradient(ellipse_70%_70%_at_50%_45%,#f6ded4_0%,#e8c3b6_45%,#d8ab9c_100%)]"
+                      : "bg-[radial-gradient(ellipse_70%_70%_at_50%_45%,#a51d1d_0%,#6d1010_45%,#2e0505_100%)]"
+                  }
+                `}
+                style={{ animationDelay: `${index * 0.3}s` }}
+              />
+              <div
+                className={`
+                  min-h-[clamp(48px,16cqh,84px)]
+                  bg-[linear-gradient(to_bottom,#f2d888_0%,#dbb14b_38%,#b0821c_74%,#8b6615_100%)]
+                `}
+              />
+            </div>
+          ))}
+        </div>
+      );
+    }
+
+    if (candidates.length === 0) {
+      return (
+        <div
+          className={`
+            flex
+            h-full
+            w-full
+            items-center
+            justify-center
+            px-2
+            text-center
+            text-[clamp(11px,1.3vw,14px)]
+            ${lightMode ? "text-[#7a5a12]" : "text-[#f0d689]"}
+          `}
+        >
+          No final interview applicants currently listed.
+        </div>
+      );
+    }
+
+    const rows = chunkIntoRows(candidates, 2);
+
+    return (
+      <div className="relative h-full w-full overflow-hidden [container-type:size]">
+        <div
+          className="absolute inset-x-0 top-0 flex flex-col gap-[2cqh]"
+          style={{ animation: `carousel-scroll ${carouselDuration}s linear infinite` }}
+        >
+          {[0, 1].map((copy) => (
+            <div key={copy} className="flex flex-col gap-[2cqh]">
+              {rows.map((row, rowIndex) => (
+                <div
+                  key={rowIndex}
+                  className="grid shrink-0 grid-cols-2 gap-[2cqw]"
+                >
+                  {row.map((candidate, cellIndex) =>
+                    renderCandidateCard(
+                      candidate,
+                      `${copy}-${rowIndex}-${cellIndex}`,
+                      row.length === 1
+                    )
+                  )}
+                </div>
+              ))}
+            </div>
+          ))}
+        </div>
+      </div>
+    );
+  }
+
+  return (
+    <main
+      className={`
+        relative
+        grid
+        h-dvh
+        place-items-center
+        overflow-hidden
+        ${lightMode ? "bg-[#f4efed]" : "bg-[#0b0102]"}
+      `}
+    >
+
+      <video
+        className={`
+          pointer-events-none
+          absolute
+          inset-0
+          z-0
+          h-full
+          w-full
+          object-cover
+          ${lightMode ? "opacity-35" : "opacity-75"}
+        `}
+        src="/visual/bg.mp4"
+        autoPlay
+        loop
+        muted
+        playsInline
+        aria-hidden="true"
+      />
+
+      <div
+        aria-hidden
+        className={`
+          pointer-events-none
+          absolute
+          inset-0
+          z-[1]
+          ${lightMode ? "bg-[#fffaf7]/70" : "bg-black/25"}
+        `}
+      />
+
+      <FullscreenSwitch />
+
+      {/* Falling embers */}
+
+      {SPARK_LAYERS.map(
+        (layer, i) =>
+          [0, -1000].map((top) => (
+            <div
+              key={`sparks-${i}-${top}`}
+              aria-hidden
+              className={`
+                pointer-events-none
+                absolute
+                left-1/2
+                -ml-[800px]
+                h-px
+                w-px
+                rounded-full
+                motion-reduce:animate-none
+                ${layer.animation}
+              `}
+              style={{
+                top,
+                boxShadow:
+                  layer.shadow,
+              }}
+            />
+          ))
+      )}
+
+      {/*
+        The board itself: one full-bleed surface that every element is
+        positioned against. It marks itself as the container-query basis,
+        so an element's cqw/cqh sizing is relative to the whole board
+        rather than to a column of it.
+      */}
+
+      <div
+        style={{ containerType: "size" }}
+        className="absolute inset-0 z-[2]"
+      >
+        {layout.layerOrder.map(renderLayer)}
+      </div>
+    </main>
+  );
